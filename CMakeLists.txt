cmake_minimum_required(VERSION 3.20)
project(LunaViewer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Network Widgets)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
pkg_check_modules(TURBOJPEG REQUIRED IMPORTED_TARGET libturbojpeg)
add_library(lunaviewer_protocol src/protocol/protocol.cpp)
target_include_directories(lunaviewer_protocol PUBLIC include)
target_link_libraries(lunaviewer_protocol PUBLIC OpenSSL::Crypto)
add_library(lunaviewer_transport src/transport/transport.cpp)
target_include_directories(lunaviewer_transport PUBLIC include)
target_link_libraries(lunaviewer_transport PUBLIC lunaviewer_protocol Qt6::Core Qt6::Network OpenSSL::SSL)
add_library(lunaviewer_screen src/screen/screen.cpp)
target_include_directories(lunaviewer_screen PUBLIC include)
target_link_libraries(lunaviewer_screen PUBLIC PkgConfig::ZSTD PkgConfig::TURBOJPEG OpenSSL::Crypto)
add_library(lunaviewer_ui src/ui/controller_window.cpp src/ui/input_mapping.cpp)
target_include_directories(lunaviewer_ui PUBLIC include)
target_link_libraries(lunaviewer_ui PUBLIC Qt6::Widgets)
add_library(lunaviewer_session app/session.cpp)
target_include_directories(lunaviewer_session PUBLIC app include)
target_link_libraries(lunaviewer_session PUBLIC lunaviewer_protocol lunaviewer_screen)
add_library(lunaviewer_logger app/logger.cpp)
target_include_directories(lunaviewer_logger PUBLIC app)
target_link_libraries(lunaviewer_logger PUBLIC Qt6::Core)
add_executable(lunaviewer app/main.cpp)
target_link_libraries(lunaviewer PRIVATE lunaviewer_session lunaviewer_transport lunaviewer_ui lunaviewer_logger)
include(CTest)
if(BUILD_TESTING)
    foreach(pair IN ITEMS "transport_test;lunaviewer_protocol" "transport_tls_test;lunaviewer_transport" "screen_tests;lunaviewer_screen" "ui_tests;lunaviewer_ui" "session_tests;lunaviewer_session" "ui_mailbox_tests;lunaviewer_session" "peer_version_tests;lunaviewer_session" "logger_tests;lunaviewer_logger")
        list(GET pair 0 name)
        list(GET pair 1 dependency)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}.cpp")
            add_executable(${name} tests/${name}.cpp)
            target_link_libraries(${name} PRIVATE ${dependency})
            if(name STREQUAL "transport_tls_test" OR name STREQUAL "ui_mailbox_tests" OR name STREQUAL "logger_tests")
                target_link_libraries(${name} PRIVATE Threads::Threads)
            endif()
            add_test(NAME ${name} COMMAND ${name})
            set_tests_properties(${name} PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endif()
    endforeach()
    add_test(NAME application_smoke COMMAND lunaviewer --smoke-exit 100 --log-dir "${CMAKE_CURRENT_BINARY_DIR}/test-logs/application")
    set_tests_properties(application_smoke PROPERTIES TIMEOUT 10 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
# Manual authorized connection diagnostic: reads credentials from stdin,
# never enables input, disconnects as soon as peer initialization succeeds.
add_executable(connection_check tools/connection_check.cpp)
target_link_libraries(connection_check PRIVATE lunaviewer_session lunaviewer_transport Qt6::Core)
